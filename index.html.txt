    <script>

        /* =========================================
           DADOS
        ========================================= */

        let presentes = JSON.parse(
            localStorage.getItem("presentesCasal")
        ) || [

            {
                id: 1,
                nome: "Perfume que eu quero muito",
                destinatario: "eu",
                descricao: "Aquele perfume que eu fiquei olhando na loja 👀",
                preco: "R$ 350,00",
                imagem: "",
                escolhido: false
            },

            {
                id: 2,
                nome: "Jantar romântico",
                destinatario: "ela",
                descricao: "Um jantarzinho só nós dois ❤️",
                preco: "R$ 200,00",
                imagem: "",
                escolhido: false
            }

        ];


        let filtroAtual = "todos";


        /* =========================================
           SALVAR
        ========================================= */

        function salvar() {

            localStorage.setItem(
                "presentesCasal",
                JSON.stringify(presentes)
            );

        }


        /* =========================================
           RENDERIZAR
        ========================================= */

        function renderizar() {

            const lista =
                document.getElementById("listaPresentes");

            lista.innerHTML = "";


            let filtrados = presentes.filter(presente => {

                if (filtroAtual === "todos") {
                    return true;
                }

                if (filtroAtual === "eu") {
                    return presente.destinatario === "eu";
                }

                if (filtroAtual === "ela") {
                    return presente.destinatario === "ela";
                }

                if (filtroAtual === "escolhidos") {
                    return presente.escolhido;
                }

            });


            if (filtrados.length === 0) {

                lista.innerHTML = `

                    <div class="painel"
                         style="grid-column:1/-1;text-align:center;">

                        <div style="font-size:60px;">
                            🥺
                        </div>

                        <h2>
                            Nenhum presente aqui ainda
                        </h2>

                        <p style="color:#777;">
                            Adicione alguma coisa que você gostaria de ganhar ❤️
                        </p>

                    </div>

                `;

                atualizarEstatisticas();

                return;
            }


            filtrados.forEach(presente => {

                const card =
                    document.createElement("article");

                card.className = "presente";


                let emoji =
                    presente.destinatario === "eu"
                        ? "👨"
                        : "👩";


                let imagemHTML = "";

                if (presente.imagem.trim() !== "") {

                    imagemHTML = `
                        <img
                            src="${presente.imagem}"
                            alt="${presente.nome}"
                            onerror="this.style.display='none';"
                        >
                    `;

                } else {

                    imagemHTML = "🎁";

                }


                card.innerHTML = `

                    ${presente.escolhido
                        ? `<div class="escolhido">
                            ❤️ Escolhido
                           </div>`
                        : ""
                    }

                    <div class="imagem-presente">
                        ${imagemHTML}
                    </div>


                    <div class="conteudo">

                        <span class="destinatario">
                            ${emoji}
                            ${presente.destinatario === "eu"
                                ? "Para mim"
                                : "Para ela"}
                        </span>


                        <h3>
                            ${presente.nome}
                        </h3>


                        <p class="descricao">
                            ${presente.descricao || "Sem descrição"}
                        </p>


                        <div class="preco">
                            ${presente.preco || "Preço não informado"}
                        </div>


                        <div class="acoes">

                            <button
                                class="btn-escolher"
                                onclick="alternarEscolhido(${presente.id})">

                                ${presente.escolhido
                                    ? "↩ Desmarcar"
                                    : "❤️ Escolher"}

                            </button>


                            <button
                                class="btn-excluir"
                                onclick="excluirPresente(${presente.id})">

                                🗑

                            </button>

                        </div>

                    </div>

                `;


                lista.appendChild(card);

            });


            atualizarEstatisticas();

        }


        /* =========================================
           ESTATÍSTICAS
        ========================================= */

        function atualizarEstatisticas() {

            document.getElementById("totalPresentes")
                .textContent = presentes.length;


            document.getElementById("meusPresentes")
                .textContent =
                presentes.filter(p => p.destinatario === "eu").length;


            document.getElementById("presentesDela")
                .textContent =
                presentes.filter(p => p.destinatario === "ela").length;

        }


        /* =========================================
           ADICIONAR PRESENTE
        ========================================= */

        function adicionarPresente() {

            const nome =
                document
                .getElementById("nomePresente")
                .value
                .trim();


            const destinatario =
                document
                .getElementById("destinatario")
                .value;


            const descricao =
                document
                .getElementById("descricaoPresente")
                .value
                .trim();


            const preco =
                document
                .getElementById("precoPresente")
                .value
                .trim();


            const imagem =
                document
                .getElementById("imagemPresente")
                .value
                .trim();


            if (!nome) {

                alert(
                    "Digite o nome do presente ❤️"
                );

                return;

            }


            const novoPresente = {

                id: Date.now(),

                nome: nome,

                destinatario: destinatario,

                descricao: descricao,

                preco: preco,

                imagem: imagem,

                escolhido: false

            };


            presentes.push(novoPresente);

            salvar();

            renderizar();

            limparFormulario();

            fecharModal();

        }


        /* =========================================
           ESCOLHER / DESMARCAR
        ========================================= */

        function alternarEscolhido(id) {

            const presente =
                presentes.find(
                    p => p.id === id
                );


            if (!presente) return;


            presente.escolhido =
                !presente.escolhido;


            salvar();

            renderizar();

        }


        /* =========================================
           EXCLUIR
        ========================================= */

        function excluirPresente(id) {

            const confirmar =
                confirm(
                    "Tem certeza que quer remover esse presente? 🥺"
                );


            if (!confirmar) return;


            presentes =
                presentes.filter(
                    p => p.id !== id
                );


            salvar();

            renderizar();

        }


        /* =========================================
           FILTROS
        ========================================= */

        function filtrar(tipo, botao) {

            filtroAtual = tipo;


            document
                .querySelectorAll(".filtro")
                .forEach(b => {
                    b.classList.remove("ativo");
                });


            botao.classList.add("ativo");


            renderizar();

        }


        /* =========================================
           MODAL
        ========================================= */

        function abrirModal() {

            document
                .getElementById("modal")
                .classList.add("ativo");

        }


        function fecharModal() {

            document
                .getElementById("modal")
                .classList.remove("ativo");

        }


        function limparFormulario() {

            document.getElementById("nomePresente").value = "";

            document.getElementById("descricaoPresente").value = "";

            document.getElementById("precoPresente").value = "";

            document.getElementById("imagemPresente").value = "";

        }


        /* =========================================
           SURPRESA
        ========================================= */

        function abrirSurpresa() {

            document
                .getElementById("modalSurpresa")
                .classList.add("ativo");


            document
                .getElementById("resultadoSurpresa")
                .textContent = "";

        }


        function fecharSurpresa() {

            document
                .getElementById("modalSurpresa")
                .classList.remove("ativo");

        }


        function sortearPresente() {

            if (presentes.length === 0) {

                document
                    .getElementById("resultadoSurpresa")
                    .textContent =
                    "Vocês ainda não adicionaram nenhum presente 🥺";

                return;

            }


            const indice =
                Math.floor(
                    Math.random() * presentes.length
                );


            const presente =
                presentes[indice];


            const nomePessoa =
                presente.destinatario === "eu"
                    ? "você"
                    : "ela";


            document
                .getElementById("resultadoSurpresa")
                .innerHTML = `

                    🎉 A surpresa é:

                    <br><br>

                    <strong>
                        ${presente.nome}
                    </strong>

                    <br>

                    <small>
                        Esse presente é para ${nomePessoa} ❤️
                    </small>

                `;

        }


        /* =========================================
           FECHAR MODAIS CLICANDO FORA
        ========================================= */

        window.onclick = function(event) {

            const modal =
                document.getElementById("modal");

            const modalSurpresa =
                document.getElementById("modalSurpresa");


            if (event.target === modal) {
                fecharModal();
            }


            if (event.target === modalSurpresa) {
                fecharSurpresa();
            }

        };


        /* =========================================
           INICIAR
        ========================================= */

        renderizar();

    </script>
